cmake_minimum_required(VERSION 3.20)
project(tg_vnc_core LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(TIGERVNC_SOURCE_DIR "" CACHE PATH "Read-only TigerVNC v1.15.0 checkout")
set(VNC_DEPS_PREFIX "" CACHE PATH "Target dependency installation prefix")
set(VNC_PIXMAN_PREFIX "" CACHE PATH "Optional separate pixman installation prefix")
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${TIGERVNC_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT revision STREQUAL "8bdf8734ce12c8ed3c232496fc8a4ccdb04636b8")
  message(FATAL_ERROR "Expected TigerVNC v1.15.0 pinned revision")
endif()
file(STRINGS "${TIGERVNC_SOURCE_DIR}/CMakeLists.txt" version_line REGEX "set\\(VERSION 1\\.15\\.0\\)")
if(NOT version_line)
  message(FATAL_ERROR "Expected VERSION 1.15.0")
endif()
if(CMAKE_CROSSCOMPILING)
  if(NOT VNC_DEPS_PREFIX)
    message(FATAL_ERROR "Cross builds require VNC_DEPS_PREFIX")
  endif()
  set(ENV{PKG_CONFIG_PATH} "")
  set(ENV{PKG_CONFIG_LIBDIR} "${VNC_DEPS_PREFIX}/lib/pkgconfig:${VNC_DEPS_PREFIX}/share/pkgconfig")
  set(PKG_CONFIG_USE_CMAKE_PREFIX_PATH FALSE)
  set(CMAKE_FIND_ROOT_PATH_MODE_LIBRARY ONLY)
  set(CMAKE_FIND_ROOT_PATH_MODE_INCLUDE ONLY)
  set(CMAKE_FIND_ROOT_PATH_MODE_PACKAGE ONLY)
  list(APPEND CMAKE_FIND_ROOT_PATH "${VNC_DEPS_PREFIX}")
endif()
if(VNC_DEPS_PREFIX)
  if(NOT IS_DIRECTORY "${VNC_DEPS_PREFIX}")
    message(FATAL_ERROR "VNC_DEPS_PREFIX does not exist: ${VNC_DEPS_PREFIX}")
  endif()
  # Explicit prefixes isolate host builds too; never fall back to system .pc files.
  set(ENV{PKG_CONFIG_PATH} "")
  set(ENV{PKG_CONFIG_LIBDIR} "${VNC_DEPS_PREFIX}/lib/pkgconfig:${VNC_DEPS_PREFIX}/share/pkgconfig")
  set(PKG_CONFIG_USE_CMAKE_PREFIX_PATH FALSE)
  list(PREPEND CMAKE_PREFIX_PATH "${VNC_DEPS_PREFIX}")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET libjpeg zlib nettle hogweed gmp)
if(VNC_PIXMAN_PREFIX)
  # Discard cached/default selections when an explicit installation is requested.
  unset(PIXMAN_INCLUDE_DIR CACHE)
  unset(PIXMAN_LIBRARY CACHE)
  unset(PIXMAN_INCLUDE_DIR)
  unset(PIXMAN_LIBRARY)
  find_path(PIXMAN_INCLUDE_DIR pixman.h
    PATHS "${VNC_PIXMAN_PREFIX}/include/pixman-1"
    NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
  find_library(PIXMAN_LIBRARY NAMES pixman-1 libpixman-1.so.0
    PATHS "${VNC_PIXMAN_PREFIX}/lib" "${VNC_PIXMAN_PREFIX}/lib64"
    NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
else()
  find_path(PIXMAN_INCLUDE_DIR pixman.h PATH_SUFFIXES pixman-1 REQUIRED)
  find_library(PIXMAN_LIBRARY NAMES pixman-1 libpixman-1.so.0 REQUIRED)
endif()
if(VNC_DEPS_PREFIX)
  file(REAL_PATH "${VNC_DEPS_PREFIX}" target_prefix)
  foreach(path IN LISTS DEPS_INCLUDE_DIRS DEPS_LIBRARY_DIRS DEPS_LINK_LIBRARIES)
    if(NOT IS_ABSOLUTE "${path}")
      continue()
    endif()
    file(REAL_PATH "${path}" resolved_path)
    cmake_path(IS_PREFIX target_prefix "${resolved_path}" NORMALIZE inside_prefix)
    if(NOT inside_prefix)
      message(FATAL_ERROR "Target pkg-config metadata escaped VNC_DEPS_PREFIX: ${path}")
    endif()
  endforeach()
endif()
if(VNC_PIXMAN_PREFIX OR CMAKE_CROSSCOMPILING)
  if(VNC_PIXMAN_PREFIX)
    file(REAL_PATH "${VNC_PIXMAN_PREFIX}" pixman_prefix)
  else()
    file(REAL_PATH "${VNC_DEPS_PREFIX}" pixman_prefix)
  endif()
  foreach(path "${PIXMAN_INCLUDE_DIR}" "${PIXMAN_LIBRARY}")
    file(REAL_PATH "${path}" resolved_path)
    cmake_path(IS_PREFIX pixman_prefix "${resolved_path}" NORMALIZE inside_prefix)
    if(NOT inside_prefix)
      message(FATAL_ERROR "Pixman dependency escaped its installation prefix (${pixman_prefix}): ${path}")
    endif()
  endforeach()
endif()
include(CTest)
find_package(Threads REQUIRED)
set(common "${TIGERVNC_SOURCE_DIR}/common")
set(os_sources Mutex.cxx Thread.cxx os.cxx)
set(network_sources Socket.cxx TcpSocket.cxx)
set(rdr_sources AESInStream.cxx AESOutStream.cxx BufferedInStream.cxx BufferedOutStream.cxx Exception.cxx FdInStream.cxx FdOutStream.cxx FileInStream.cxx HexInStream.cxx HexOutStream.cxx RandomStream.cxx ZlibInStream.cxx ZlibOutStream.cxx)
set(rfb_sources KeysymStr.c Region.cxx CConnection.cxx CMsgHandler.cxx CMsgReader.cxx CMsgWriter.cxx CSecurityPlain.cxx CSecurityStack.cxx CSecurityVeNCrypt.cxx CSecurityVncAuth.cxx CSecurityDH.cxx CSecurityMSLogonII.cxx CSecurityRSAAES.cxx Configuration.cxx CopyRectDecoder.cxx Cursor.cxx DecodeManager.cxx Decoder.cxx d3des.c HextileDecoder.cxx JpegDecompressor.cxx LogWriter.cxx Logger.cxx Logger_file.cxx Logger_stdio.cxx PixelBuffer.cxx PixelFormat.cxx RREDecoder.cxx RawDecoder.cxx ServerParams.cxx Security.cxx SecurityClient.cxx Timer.cxx TightDecoder.cxx ZRLEDecoder.cxx encodings.cxx util.cxx)
set(sources "")
foreach(group os network rdr rfb)
  foreach(source IN LISTS ${group}_sources)
    list(APPEND sources "${common}/${group}/${source}")
  endforeach()
endforeach()
# Always refresh the build-only copy before applying the exact patch, so
# reconfiguration is idempotent and upstream remains untouched.
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS" OR (BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING))
  find_program(PATCH_EXECUTABLE patch REQUIRED)
  set(patched_thread "${CMAKE_CURRENT_BINARY_DIR}/ohos-adaptation/Thread.cxx")
  configure_file("${common}/os/Thread.cxx" "${patched_thread}" COPYONLY)
  execute_process(COMMAND "${PATCH_EXECUTABLE}" --batch --fuzz=0
    "${patched_thread}" "${CMAKE_CURRENT_SOURCE_DIR}/ohos-thread.patch"
    COMMAND_ERROR_IS_FATAL ANY)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/ohos-thread.patch")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  list(REMOVE_ITEM sources "${common}/os/Thread.cxx")
  list(APPEND sources "${patched_thread}")
endif()
add_library(tigervnc_client STATIC ${sources})
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  target_compile_definitions(tigervnc_client PRIVATE TG_OHOS_NO_PTHREAD_CANCEL)
endif()
target_include_directories(tigervnc_client PUBLIC "${common}" PRIVATE "${PIXMAN_INCLUDE_DIR}")
target_compile_definitions(tigervnc_client PUBLIC HAVE_NETTLE HAVE_SYS_SELECT_H)
target_link_libraries(tigervnc_client PUBLIC PkgConfig::DEPS Threads::Threads "${PIXMAN_LIBRARY}")
add_executable(vnc-core main.cxx)
target_link_libraries(vnc-core PRIVATE tigervnc_client)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/upstream-sources.txt" CONTENT "$<JOIN:$<TARGET_PROPERTY:tigervnc_client,SOURCES>,\n>\n")
if(BUILD_TESTING AND (CMAKE_SYSTEM_NAME STREQUAL "OHOS" OR NOT CMAKE_CROSSCOMPILING))
  add_executable(thread-lifecycle thread-lifecycle.cxx "${patched_thread}"
    "${common}/os/Mutex.cxx" "${common}/rdr/Exception.cxx")
  target_include_directories(thread-lifecycle PRIVATE "${common}")
  target_compile_definitions(thread-lifecycle PRIVATE TG_OHOS_NO_PTHREAD_CANCEL)
  target_link_libraries(thread-lifecycle PRIVATE tigervnc_client)
endif()
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME ohos_thread_lifecycle COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/thread-lifecycle.py" "$<TARGET_FILE:thread-lifecycle>")
  set_tests_properties(ohos_thread_lifecycle PROPERTIES TIMEOUT 8)
  # Exercise real DecodeThread teardown through existing fixtures with the
  # same patched Thread implementation; the official host target stays original.
  add_library(tigervnc_client_ohos_test OBJECT "${patched_thread}")
  target_include_directories(tigervnc_client_ohos_test PRIVATE "${common}")
  target_compile_definitions(tigervnc_client_ohos_test PRIVATE TG_OHOS_NO_PTHREAD_CANCEL)
  target_link_libraries(tigervnc_client_ohos_test PUBLIC tigervnc_client)
  add_executable(vnc-core-ohos-test main.cxx)
  target_link_libraries(vnc-core-ohos-test PRIVATE tigervnc_client_ohos_test)
  foreach(case raw truncated)
    add_test(NAME ohos_adapted_${case} COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/fixture.py" "$<TARGET_FILE:vnc-core-ohos-test>" ${case})
    set_tests_properties(ohos_adapted_${case} PROPERTIES TIMEOUT 8)
  endforeach()
  add_test(NAME ra2_256_factory COMMAND vnc-core --capabilities)
  foreach(case raw truncated unsupported none_rejected timeout)
    add_test(NAME ${case} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/fixture.py" "$<TARGET_FILE:vnc-core>" ${case})
    set_tests_properties(${case} PROPERTIES TIMEOUT 8)
  endforeach()
  set_tests_properties(ra2_256_factory PROPERTIES TIMEOUT 5)
endif()
